rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn() &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isProtectedUserEmail(email) {
      return email in ['kevlimpens@gmail.com', 'vragen@scheikundeles.nl'];
    }

    // Gebruikers documenten
    match /users/{userId} {
      allow read: if signedIn() && (
        request.auth.uid == userId ||
        isAdmin()
      );
      allow create, update: if signedIn() && request.auth.uid == userId;
      allow delete: if isAdmin() &&
        resource.data.role == 'student' &&
        !isProtectedUserEmail(resource.data.email);
    }

    // Antwoorden - leerlingen schrijven hun eigen, admins overal
    match /userAnswers/{docId} {
      allow read: if request.auth != null && (
        resource.data.userId == request.auth.uid ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'
      );
      allow write: if request.auth != null && (
        request.auth.uid == resource.data.userId ||
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin'
      );
    }

    // CMS Collections - PERMISSIEF FOR TESTING
    match /vakken/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    match /vak/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    match /leerjaar/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    match /niveau/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    match /hoofdstuk/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    match /paragraaf/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    match /vraag/{document=**} {
      allow read, write, delete: if request.auth != null;
    }

    // Klassen (classes)
    match /klassen/{klasId} {
      allow read: if signedIn();
      allow write, delete: if isAdmin();
    }

    // Voortgang (student progress)
    match /voortgang/{document=**} {
      allow read: if signedIn() && (
        resource.data.userId == request.auth.uid ||
        isAdmin()
      );
      allow create, update: if signedIn() &&
        request.resource.data.userId == request.auth.uid;
      allow delete: if isAdmin();
    }

    // Door docenten/admins afgevinkte voortgangssignalen.
    match /progressSignalAcknowledgements/{signalId} {
      allow read, create, update, delete: if isAdmin();
    }

    // Tijdelijke leerlingen uit foto-import
    match /pendingStudents/{document=**} {
      allow read, write, delete: if isAdmin();
    }

    // Leerlingfoto-import: tijdelijke importmetadata en croprecords.
    // De definitieve koppeling naar users/{uid}.photo loopt via Cloud Functions.
    match /photoImports/{importId} {
      allow read, create, update, delete: if isAdmin();

      match /crops/{cropId} {
        allow read, create, update, delete: if isAdmin();
      }
    }

    // Crop metadata
    match /questionMetadata/{paragraphId}/questions/{questionId} {
      allow read, write: if request.auth != null;
    }
  }
}
